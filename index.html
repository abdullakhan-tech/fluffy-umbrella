async function askZaythar(message) {
  const response = await fetch('/api/chat', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      message,
      language: 'en-US'
    })
  });

  if (!response.ok) {
    throw new Error('AI service unavailable');
  }
  return response.json();
}

A secure API request pattern. Keep provider credentials on your backend, never in browser code.
